\begin{document}
\definecolor{tensix}{HTML}{FCE9D4}
\definecolor{dram}{HTML}{BFEFE4}
\definecolor{eth}{HTML}{D6E0FA}
\definecolor{other}{HTML}{EEEEEE}
\definecolor{rd}{HTML}{2A78D6}
\definecolor{wr}{HTML}{D6452A}
\def\s{0.4}

\newcommand{\grid}{%
  \foreach \x in {0,...,9} \foreach \y in {0,...,11} {%
    \def\c{tensix}%
    \ifnum\y=0 \def\c{eth}\fi
    \ifnum\y=6 \def\c{eth}\fi
    \ifnum\x=0 \def\c{other}\fi
    \ifnum\x=0 \ifnum\y=0 \def\c{dram}\fi\ifnum\y=1 \def\c{dram}\fi
      \ifnum\y=5 \def\c{dram}\fi\ifnum\y=6 \def\c{dram}\fi
      \ifnum\y=7 \def\c{dram}\fi\ifnum\y=11 \def\c{dram}\fi\fi
    \ifnum\x=5 \def\c{dram}\fi
    \fill[\c, draw=black!40, line width=0.2pt]
      (\x*\s+0.03, -\y*\s-0.03) rectangle (\x*\s+\s-0.03, -\y*\s-\s+0.03);
  }%
}
% centre of tile (x,y), shifted by (dx,dy)
\newcommand{\tc}[4]{({#1*\s+\s/2+#3}, {-#2*\s-\s/2+#4})}
\tikzset{route/.style={line width=1.1pt, -{Stealth[length=3pt]}, rounded corners=1pt}}

\begin{tikzpicture}
  % (a) reads on NoC 1, writes on NoC 0
  \begin{scope}
    \grid
    \node[font=\scriptsize, anchor=south] at (2,0.05) {(a) reads NoC\,1, writes NoC\,0};
    % read: DRAM (5,7) -> core (2,2), NoC 1: up (Y) first, then left
    \draw[route, rd] \tc{5}{7}{-0.06}{0} -- \tc{5}{2}{-0.06}{0.06} -- \tc{2}{2}{0.06}{0.06};
    % write: core (2,2) -> DRAM (5,7), NoC 0: right (X) first, then down
    \draw[route, wr] \tc{2}{2}{0.06}{-0.06} -- \tc{5}{2}{0.06}{-0.06} -- \tc{5}{7}{0.06}{0};
    \draw[black, line width=0.6pt] \tc{2}{2}{0}{0} circle (0.13);
    \draw[black, line width=0.6pt] \tc{5}{7}{0}{0} circle (0.13);
  \end{scope}
  % (b) reads on NoC 0, writes on NoC 1
  \begin{scope}[xshift=4.4cm]
    \grid
    \node[font=\scriptsize, anchor=south] at (2,0.05) {(b) reads NoC\,0, writes NoC\,1};
    % read: DRAM (5,4) -> core (8,9), NoC 0: right (X) first, then down
    \draw[route, rd] \tc{5}{4}{0}{0.06} -- \tc{8}{4}{0.06}{0.06} -- \tc{8}{9}{0.06}{0};
    % write: core (8,9) -> DRAM (5,4), NoC 1: up (Y) first, then left
    \draw[route, wr] \tc{8}{9}{-0.06}{0} -- \tc{8}{4}{-0.06}{-0.06} -- \tc{5}{4}{0}{-0.06};
    \draw[black, line width=0.6pt] \tc{8}{9}{0}{0} circle (0.13);
    \draw[black, line width=0.6pt] \tc{5}{4}{0}{0} circle (0.13);
  \end{scope}
  % legend
  \begin{scope}[yshift=-5.25cm, font=\scriptsize]
    \fill[tensix, draw=black!40] (0,0) rectangle (0.25,0.25); \node[anchor=west] at (0.3,0.125) {Tensix};
    \fill[dram, draw=black!40] (1.4,0) rectangle (1.65,0.25); \node[anchor=west] at (1.7,0.125) {DRAM};
    \fill[eth, draw=black!40] (2.8,0) rectangle (3.05,0.25); \node[anchor=west] at (3.1,0.125) {Ethernet};
    \draw[route, rd] (4.5,0.125) -- (5.0,0.125); \node[anchor=west] at (5.05,0.125) {read data};
    \draw[route, wr] (6.5,0.125) -- (7.0,0.125); \node[anchor=west] at (7.05,0.125) {write data};
  \end{scope}
\end{tikzpicture}
